cmake_minimum_required(VERSION 3.16)
project(uhg_paper_baselines LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(VSAG_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(VSAG_BUILD_ROOT "${VSAG_REPO}/build-release" CACHE PATH "VSAG build containing the production SIMD library")
set(HDF5_ROOT "${VSAG_REPO}/build-release/hdf5/install" CACHE PATH "HDF5 installation")
set(OPENBLAS_ROOT "${VSAG_REPO}/build-release/openblas/install" CACHE PATH "OpenBLAS installation")
find_path(PAPER_HDF5_INCLUDE H5Cpp.h HINTS "${HDF5_ROOT}/include" REQUIRED)
find_library(PAPER_HDF5_CPP hdf5_cpp HINTS "${HDF5_ROOT}/lib" REQUIRED)
find_library(PAPER_HDF5_C hdf5 HINTS "${HDF5_ROOT}/lib" REQUIRED)
find_path(PAPER_BLAS_INCLUDE cblas.h HINTS "${OPENBLAS_ROOT}/include" REQUIRED)
find_library(PAPER_BLAS openblas HINTS "${OPENBLAS_ROOT}/lib" REQUIRED)
find_path(PAPER_NLOHMANN_INCLUDE nlohmann/json.hpp
    HINTS "${VSAG_BUILD_ROOT}/_deps/nlohmann_json-src/include" REQUIRED)
find_library(PAPER_VSAG vsag HINTS "${VSAG_BUILD_ROOT}/src" NO_DEFAULT_PATH REQUIRED)
find_package(OpenMP REQUIRED)

if(TARGET simd)
    set(PAPER_VSAG_SIMD simd)
else()
    find_library(PAPER_VSAG_SIMD simd HINTS "${VSAG_BUILD_ROOT}/src/simd" NO_DEFAULT_PATH REQUIRED)
endif()
if(TARGET cpuinfo)
    set(PAPER_CPUINFO cpuinfo)
else()
    find_library(PAPER_CPUINFO cpuinfo HINTS "${VSAG_BUILD_ROOT}/_deps/cpuinfo-build" NO_DEFAULT_PATH REQUIRED)
endif()

function(configure_paper_baseline_target target)
    target_include_directories(${target} PRIVATE
        "${VSAG_REPO}/src" "${VSAG_REPO}/include" "${PAPER_BLAS_INCLUDE}"
        "${PAPER_NLOHMANN_INCLUDE}")
    target_link_libraries(${target} PRIVATE
        "${PAPER_BLAS}" OpenMP::OpenMP_CXX ${PAPER_VSAG_SIMD} ${PAPER_CPUINFO}
        "${PAPER_VSAG}" m)
    target_compile_options(${target} PRIVATE -Ofast -O3 -Wall -Wextra -Werror)
    target_link_options(${target} PRIVATE -static-libstdc++)
endfunction()

add_executable(uhg_paper_baselines main.cpp)
configure_paper_baseline_target(uhg_paper_baselines)
target_include_directories(uhg_paper_baselines PRIVATE "${PAPER_HDF5_INCLUDE}")
target_link_libraries(uhg_paper_baselines PRIVATE "${PAPER_HDF5_CPP}" "${PAPER_HDF5_C}")
enable_testing()
add_executable(paper_baselines_test tests.cpp)
configure_paper_baseline_target(paper_baselines_test)
add_test(NAME paper_baselines_correctness COMMAND paper_baselines_test)

add_executable(uhg_extra_baselines extra_main.cpp)
configure_paper_baseline_target(uhg_extra_baselines)
target_include_directories(uhg_extra_baselines SYSTEM PRIVATE "${VSAG_REPO}/include")
target_include_directories(uhg_extra_baselines PRIVATE "${PAPER_HDF5_INCLUDE}")
target_link_libraries(uhg_extra_baselines PRIVATE "${PAPER_HDF5_CPP}" "${PAPER_HDF5_C}")
add_executable(extra_baselines_test extra_tests.cpp)
configure_paper_baseline_target(extra_baselines_test)
add_test(NAME extra_baselines_correctness COMMAND extra_baselines_test)
